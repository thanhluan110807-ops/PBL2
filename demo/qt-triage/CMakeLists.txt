# Chỉ được gọi khi PBL2_BUILD_QT_DEMO=ON. Qt không đi vào pbl2_core.
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
message(STATUS "Native demo: Qt ${QT_VERSION_MAJOR}, Widgets only")

add_library(pbl2_demo_state STATIC model/DemoTriageService.cpp model/DemoSeed.cpp)
target_include_directories(pbl2_demo_state PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(pbl2_demo_state PUBLIC pbl2_build_options)
if(MSVC)
    target_compile_options(pbl2_demo_state PUBLIC /utf-8)
endif()

add_library(pbl2_demo_ui STATIC ui/DemoWindow.cpp ui/DemoDialogs.cpp ui/DemoStyle.cpp)
target_link_libraries(pbl2_demo_ui PUBLIC pbl2_demo_state Qt${QT_VERSION_MAJOR}::Widgets)
add_executable(pbl2_qt_demo main.cpp)
target_link_libraries(pbl2_qt_demo PRIVATE pbl2_demo_ui)

if(BUILD_TESTING)
    add_executable(test_demo_state tests/test_demo_state.cpp)
    target_link_libraries(test_demo_state PRIVATE pbl2_demo_state)
    add_executable(test_demo_widgets tests/test_demo_widgets.cpp)
    target_link_libraries(test_demo_widgets PRIVATE pbl2_demo_ui)
    foreach(Ten IN ITEMS test_demo_state test_demo_widgets)
        if(MSVC)
            target_compile_options(${Ten} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${Ten} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${Ten} COMMAND ${Ten})
        set_tests_properties(${Ten} PROPERTIES TIMEOUT 30)
    endforeach()
    set_tests_properties(test_demo_state PROPERTIES ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
    set_tests_properties(test_demo_widgets PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
endif()
